  <script>
    const {
      classifyUrl,
      decodeTimelinePayload,
      getDurationHintSeconds,
      getPlaybackStartIndex,
      parseNodeText
    } = window.ProgramMonitorUtils || {};

    const elements = {
      baseVideo: document.getElementById("baseVideo"),
      baseFrame: document.getElementById("baseFrame"),
      overlayLayer: document.getElementById("overlayLayer"),
      statNode: document.getElementById("statNode"),
      statTotal: document.getElementById("statTotal"),
      statT: document.getElementById("statT"),
      statDur: document.getElementById("statDur"),
      stageMeta: document.getElementById("stageMeta"),
      stageStatus: document.getElementById("stageStatus")
    };

    const state = {
      nodes: [],
      activeIndex: 0,
      playing: false,
      stopRequested: false
    };

    let overlayVideos = [];
    let overlayAudios = [];
    let rafId = null;
    let baseEndedHandler = null;
    let activeBaseKind = "video";
    let baseStartTime = 0;

    function setStatus(text) {
      elements.stageStatus.textContent = text || "";
    }

    function setBaseKind(kind, url) {
      activeBaseKind = kind;
      if (kind === "page") {
        elements.baseVideo.pause();
        elements.baseVideo.removeAttribute("src");
        elements.baseVideo.style.display = "none";
        elements.baseFrame.style.display = "block";
        elements.baseFrame.src = url || "about:blank";
      } else {
        elements.baseFrame.removeAttribute("src");
        elements.baseFrame.style.display = "none";
        elements.baseVideo.style.display = "block";
      }
    }

    function clearLayers() {
      overlayVideos.forEach((video) => video.pause());
      overlayAudios.forEach((audio) => audio.pause());
      overlayVideos = [];
      overlayAudios = [];
      elements.overlayLayer.innerHTML = "";
    }

    function clearBaseHandlers() {
      if (baseEndedHandler) {
        elements.baseVideo.removeEventListener("ended", baseEndedHandler);
        baseEndedHandler = null;
      }
    }

    function loadVideoMeta(videoEl, url) {
      return new Promise((resolve, reject) => {
        const onLoaded = () => {
          cleanup();
          resolve(videoEl.duration || 0);
        };
        const onErr = () => {
          cleanup();
          reject(new Error(`Failed to load video metadata: ${url}`));
        };
        const cleanup = () => {
          videoEl.removeEventListener("loadedmetadata", onLoaded);
          videoEl.removeEventListener("error", onErr);
        };

        videoEl.addEventListener("loadedmetadata", onLoaded, { once: true });
        videoEl.addEventListener("error", onErr, { once: true });
        videoEl.src = url;
        videoEl.load();
      });
    }

    function resolveDurationInfo({ baseKind, baseUrl, overrideSeconds, baseDuration, durationHintSeconds }) {
      const override = Number.isFinite(overrideSeconds) && overrideSeconds > 0 ? overrideSeconds : 0;
      const base = Number.isFinite(baseDuration) && baseDuration > 0 ? baseDuration : 0;
      const hintValue = Number.isFinite(durationHintSeconds) ? durationHintSeconds : getDurationHintSeconds(baseUrl);
      const hint = Number.isFinite(hintValue) && hintValue > 0 ? hintValue : 0;

      if (baseKind === "page") {
        if (override) {
          return { duration: override, source: "override" };
        }
        if (hint) {
          return { duration: hint, source: "hint" };
        }
        return { duration: 0, source: "none" };
      }

      if (override) {
        return { duration: override, source: "override" };
      }
      if (base) {
        return { duration: base, source: "base" };
      }
      if (hint) {
        return { duration: hint, source: "hint" };
      }
      return { duration: 0, source: "none" };
    }

    async function primeNode(index) {
      clearLayers();
      clearBaseHandlers();

      const node = state.nodes[index];
      if (!node) {
        setBaseKind("video");
        elements.baseVideo.removeAttribute("src");
        elements.baseFrame.removeAttribute("src");
        elements.statDur.textContent = "0.00";
        return;
      }

      const parsed = parseNodeText(node.text);
      const overrideSeconds = Number(node.durationOverride);
      const baseKind = classifyUrl(parsed.baseUrl);
      const durationHintSeconds = getDurationHintSeconds(parsed.baseUrl);

      if (!parsed.baseUrl) {
        setBaseKind("video");
        elements.baseVideo.removeAttribute("src");
        elements.baseFrame.removeAttribute("src");
        elements.statDur.textContent = "0.00";
        return;
      }

      if (baseKind === "page") {
        setBaseKind("page", parsed.baseUrl);
      } else {
        setBaseKind("video", parsed.baseUrl);
      }

      let duration = 0;
      if (baseKind !== "page") {
        elements.baseVideo.loop = false;
        elements.baseVideo.muted = false;
        elements.baseVideo.playsInline = true;
        elements.baseVideo.preload = "metadata";

        try {
          duration = await loadVideoMeta(elements.baseVideo, parsed.baseUrl);
        } catch (error) {
          duration = 0;
        }
      }

      duration = resolveDurationInfo({
        baseKind,
        baseUrl: parsed.baseUrl,
        overrideSeconds,
        baseDuration: duration,
        durationHintSeconds
      }).duration;

      elements.statDur.textContent = duration.toFixed(2);

      parsed.layers.forEach((url) => {
        const kind = classifyUrl(url);
        if (kind === "audio") {
          const audio = document.createElement("audio");
          audio.src = url;
          audio.loop = true;
          audio.preload = "auto";
          overlayAudios.push(audio);
          return;
        }

        if (kind === "image") {
          const image = document.createElement("img");
          image.src = url;
          image.alt = "";
          elements.overlayLayer.appendChild(image);
          return;
        }

        if (kind === "page") {
          const frame = document.createElement("iframe");
          frame.src = url;
          frame.title = "Overlay URL";
          frame.loading = "eager";
          frame.setAttribute("aria-hidden", "true");
          elements.overlayLayer.appendChild(frame);
          return;
        }

        const video = document.createElement("video");
        video.src = url;
        video.loop = true;
        video.muted = true;
        video.playsInline = true;
        video.preload = "auto";
        elements.overlayLayer.appendChild(video);
        overlayVideos.push(video);
      });
    }

    function stopAll() {
      state.stopRequested = true;
      state.playing = false;
      if (rafId) {
        cancelAnimationFrame(rafId);
        rafId = null;
      }
      clearBaseHandlers();
      elements.baseVideo.pause();

        video.currentTime = 0;

      elements.statNode.textContent = "--";

    function pauseAll() {
      state.playing = false;
      elements.baseVideo.pause();
      overlayVideos.forEach((video) => video.pause());
      overlayAudios.forEach((audio) => audio.pause());
    }

    async function playActive() {
      state.stopRequested = false;
      state.playing = true;
      if (rafId) {
        cancelAnimationFrame(rafId);
        rafId = null;
      }

      const startIndex = getPlaybackStartIndex(state.activeIndex, state.nodes.length);
      if (startIndex === -1) {
        state.playing = false;
        setStatus("No nodes available.");
        return;
      }

      if (startIndex !== state.activeIndex) {
        state.activeIndex = startIndex;
      }

      const index = state.activeIndex;
      const overrideSeconds = Number(state.nodes[index].durationOverride);
      elements.statNode.textContent = String(index + 1);

      await primeNode(index);

      const parsed = parseNodeText(state.nodes[index].text);
      const durationHintSeconds = getDurationHintSeconds(parsed.baseUrl);
      if (!parsed.baseUrl) {
        state.playing = false;
        setStatus("Missing base URL in this node.");
        return;
      }

      const baseKind = classifyUrl(parsed.baseUrl);
      activeBaseKind = baseKind;
      baseStartTime = performance.now();

      overlayVideos.forEach((video) => {
        video.currentTime = 0;
        video.play().catch(() => {});
      });
      overlayAudios.forEach((audio) => {
        audio.currentTime = 0;
        audio.play().catch(() => {});
      });

      if (baseKind !== "page") {
        elements.baseVideo.currentTime = 0;
        await elements.baseVideo.play().catch(() => {
          state.playing = false;
        });
      }

      let advanced = false;
      const advance = () => {
        if (advanced) {
          return;
        }
        advanced = true;

        overlayVideos.forEach((video) => video.pause());
        overlayAudios.forEach((audio) => audio.pause());
        elements.baseVideo.pause();

        if (state.stopRequested) {
          state.playing = false;
          return;
        }

        if (state.activeIndex < state.nodes.length - 1) {
          state.activeIndex += 1;
          playActive().catch(() => {});
        } else {
          state.playing = false;
          setStatus("Reached end of timeline.");
        }
      };

      clearBaseHandlers();
      if (baseKind !== "page") {
        baseEndedHandler = () => {
          if (state.playing) {
            advance();
          }
        };
        elements.baseVideo.addEventListener("ended", baseEndedHandler);
      }

      const tick = () => {
        if (!state.playing) {
          return;
        }

        const current = activeBaseKind === "page"
          ? (performance.now() - baseStartTime) / 1000
          : (elements.baseVideo.currentTime || 0);

        const durationInfo = resolveDurationInfo({
          baseKind: activeBaseKind,
          baseUrl: parsed.baseUrl,
          overrideSeconds,
          baseDuration: elements.baseVideo.duration,
          durationHintSeconds
        });

        const duration = durationInfo.duration;
        elements.statT.textContent = current.toFixed(2);
        elements.statDur.textContent = (Number.isFinite(duration) ? duration : 0).toFixed(2);

        const shouldAdvanceOnDuration = duration > 0
          && (activeBaseKind === "page" || durationInfo.source === "override");

        if (shouldAdvanceOnDuration && current >= duration - 0.05) {
          advance();
          return;
        }

        rafId = requestAnimationFrame(tick);
      };

      rafId = requestAnimationFrame(tick);
    }

    function loadTimeline() {
      const params = new URLSearchParams(window.location.search);
      const data = params.get("data");
      const name = params.get("name");
      const payload = decodeTimelinePayload ? decodeTimelinePayload(data) : null;
      let fallback = null;
      try {
        const stored = window.localStorage.getItem("program-monitor.timeline.v1");
        fallback = stored ? JSON.parse(stored) : null;
      } catch (error) {
        fallback = null;
      }
      const timeline = payload && Array.isArray(payload.nodes) ? payload : fallback;

      if (!timeline || !Array.isArray(timeline.nodes) || !timeline.nodes.length) {
        elements.stageMeta.textContent = "No timeline data provided.";
        setStatus("Missing timeline payload. Provide ?data= in the URL.");
        return;
      }

      state.nodes = timeline.nodes;
      state.activeIndex = 0;
      elements.statTotal.textContent = String(state.nodes.length);
      elements.stageMeta.textContent = name ? `Project: ${name}` : `Nodes: ${state.nodes.length}`;
    }

    document.getElementById("btnPlay").addEventListener("click", () => {
      if (!state.playing) {
        playActive().catch(() => {});
      }
    });

    document.getElementById("btnPause").addEventListener("click", () => pauseAll());
    document.getElementById("btnStop").addEventListener("click", () => stopAll());
    document.getElementById("btnRestart").addEventListener("click", () => {
      stopAll();
      state.activeIndex = 0;
      playActive().catch(() => {});
    });

    loadTimeline();

    if (state.nodes.length) {
      primeNode(0).catch(() => {});
      playActive().catch(() => {});
      setStatus("Playing timeline preview.");
    }
  </script>
</body>

    // Utility functions (placeholder - replace with your actual timeline-utils.js)
    const classifyUrl = (url) => {
      if (!url) return 'unknown';
      if (url.match(/\.(mp4|webm|mov)$/i)) return 'video';
      if (url.match(/\.(mp3|wav|ogg)$/i)) return 'audio';
      if (url.match(/\.(jpg|jpeg|png|gif|webp)$/i)) return 'image';
      if (url.startsWith('http') || url.startsWith('https')) return 'page';
      return 'unknown';
    };

    const parseNodeText = (text) => {
      const lines = (text || '').split('\n').map(l => l.trim()).filter(Boolean);
      return {
        baseUrl: lines[0] || '',
        layers: lines.slice(1)
      };
    };

    const getDurationHintSeconds = (url) => {
      const match = url?.match(/[?&]t=(\d+)/);
      return match ? parseFloat(match[1]) : 0;
    };

    const decodeTimelinePayload = (data) => {
      if (!data) return null;
      try {
        return JSON.parse(atob(data));
      } catch {
        return null;
      }
    };

    const getPlaybackStartIndex = (current, total) => {
      return total > 0 ? 0 : -1;
    };

    // State
    const state = {
      nodes: [],
      activeIndex: 0,
      playing: false,
      stopRequested: false,
      totalDuration: 0,
      currentTime: 0
    };

    const elements = {
      baseVideo: document.getElementById('baseVideo'),
      baseFrame: document.getElementById('baseFrame'),
      overlayLayer: document.getElementById('overlayLayer'),
      projectName: document.getElementById('projectName'),
      projectMeta: document.getElementById('projectMeta'),
      statusText: document.getElementById('statusText'),
      currentTime: document.getElementById('currentTime'),
      totalTime: document.getElementById('totalTime'),
      timelineTrack: document.getElementById('timelineTrack'),
      timelineProgress: document.getElementById('timelineProgress'),
      playhead: document.getElementById('playhead'),
      nodeContainer: document.getElementById('nodeContainer'),
      nodeTrack: document.getElementById('nodeTrack')
    };

    let overlayVideos = [];
    let overlayAudios = [];
    let rafId = null;
    let activeBaseKind = 'video';
    let baseStartTime = 0;

    // Format time as MM:SS.s
    function formatTime(seconds) {
      const mins = Math.floor(seconds / 60);
      const secs = seconds % 60;
      return `${String(mins).padStart(2, '0')}:${secs.toFixed(1).padStart(4, '0')}`;
    }

    // Update timeline progress
    function updateTimelineUI() {
      const progress = state.totalDuration > 0 ? (state.currentTime / state.totalDuration) * 100 : 0;
      elements.timelineProgress.style.width = `${progress}%`;
      elements.playhead.style.left = `${progress}%`;
      elements.currentTime.textContent = formatTime(state.currentTime);
      elements.totalTime.textContent = formatTime(state.totalDuration);
    }

    // Render node clips in timeline
    function renderNodeTrack() {
      elements.nodeContainer.innerHTML = '';
      
      state.nodes.forEach((node, index) => {
        const clip = document.createElement('div');
        clip.className = 'node-clip';
        if (index === state.activeIndex) {
          clip.classList.add('active');
        }
        
        clip.innerHTML = `
          <div class="node-index">${index + 1}</div>
          <div class="node-label">Node ${index + 1}</div>
          <div class="node-duration">${node.durationOverride || '?'}s</div>
        `;
        
        clip.addEventListener('click', () => {
          jumpToNode(index);
        });
        
        elements.nodeContainer.appendChild(clip);
      });

      // Scroll active node into view
      const activeClip = elements.nodeContainer.children[state.activeIndex];
      if (activeClip) {
        activeClip.scrollIntoView({ behavior: 'smooth', inline: 'center', block: 'nearest' });
      }
    }

    // Jump to specific node
    function jumpToNode(index) {
      const wasPlaying = state.playing;
      stopAll();
      state.activeIndex = index;
      renderNodeTrack();
      primeNode(index).then(() => {
        if (wasPlaying) {
          playActive();
        }
      });
    }

    // Set base media type
    function setBaseKind(kind, url) {
      activeBaseKind = kind;
      elements.baseVideo.classList.remove('active');
      elements.baseFrame.classList.remove('active');
      
      if (kind === 'page') {
        elements.baseFrame.classList.add('active');
        elements.baseFrame.src = url || 'about:blank';
      } else {
        elements.baseVideo.classList.add('active');
      }
    }

    // Clear overlay layers
    function clearLayers() {
      overlayVideos.forEach(v => v.pause());
      overlayAudios.forEach(a => a.pause());
      overlayVideos = [];
      overlayAudios = [];
      elements.overlayLayer.innerHTML = '';
    }

    // Load video metadata
    function loadVideoMeta(videoEl, url) {
      return new Promise((resolve) => {
        const onLoaded = () => {
          cleanup();
          resolve(videoEl.duration || 0);
        };
        const onErr = () => {
          cleanup();
          resolve(0);
        };
        const cleanup = () => {
          videoEl.removeEventListener('loadedmetadata', onLoaded);
          videoEl.removeEventListener('error', onErr);
        };

        videoEl.addEventListener('loadedmetadata', onLoaded, { once: true });
        videoEl.addEventListener('error', onErr, { once: true });
        videoEl.src = url;
        videoEl.load();
      });
    }

    // Prime a node for playback
    async function primeNode(index) {
      clearLayers();
      
      const node = state.nodes[index];
      if (!node) {
        setBaseKind('video');
        return;
      }

      const parsed = parseNodeText(node.text);
      const baseKind = classifyUrl(parsed.baseUrl);
      
      if (!parsed.baseUrl) {
        setBaseKind('video');
        return;
      }

      setBaseKind(baseKind, parsed.baseUrl);

      let duration = parseFloat(node.durationOverride) || 0;
      
      if (baseKind === 'video' && !duration) {
        duration = await loadVideoMeta(elements.baseVideo, parsed.baseUrl);
      }

      // Add overlay layers
      parsed.layers.forEach(url => {
        const kind = classifyUrl(url);
        
        if (kind === 'audio') {
          const audio = document.createElement('audio');
          audio.src = url;
          audio.loop = true;
          overlayAudios.push(audio);
        } else if (kind === 'image') {
          const img = document.createElement('img');
          img.src = url;
          elements.overlayLayer.appendChild(img);
        } else if (kind === 'video') {
          const video = document.createElement('video');
          video.src = url;
          video.loop = true;
          video.muted = true;
          video.playsInline = true;
          elements.overlayLayer.appendChild(video);
          overlayVideos.push(video);
        } else if (kind === 'page') {
          const iframe = document.createElement('iframe');
          iframe.src = url;
          elements.overlayLayer.appendChild(iframe);
        }
      });

      return duration;
    }

    // Calculate total timeline duration
    function calculateTotalDuration() {
      state.totalDuration = state.nodes.reduce((sum, node) => {
        return sum + (parseFloat(node.durationOverride) || 5);
      }, 0);
    }

    // Play active node
    async function playActive() {
      state.playing = true;
      state.stopRequested = false;
      
      const index = state.activeIndex;
      const node = state.nodes[index];
      
      if (!node) {
        state.playing = false;
        return;
      }

      renderNodeTrack();
      
      const duration = await primeNode(index);
      const parsed = parseNodeText(node.text);
      const baseKind = classifyUrl(parsed.baseUrl);
      
      baseStartTime = performance.now();
      
      // Start overlays
      overlayVideos.forEach(v => {
        v.currentTime = 0;
        v.play().catch(() => {});
      });
      overlayAudios.forEach(a => {
        a.currentTime = 0;
        a.play().catch(() => {});
      });

      // Start base video
      if (baseKind === 'video') {
        elements.baseVideo.currentTime = 0;
        await elements.baseVideo.play().catch(() => {});
      }

      // Animation loop
      const tick = () => {
        if (!state.playing) return;

        const elapsed = (performance.now() - baseStartTime) / 1000;
        const nodeDuration = duration || 5;
        
        // Calculate current time in timeline
        let timelineTime = 0;
        for (let i = 0; i < state.activeIndex; i++) {
          timelineTime += parseFloat(state.nodes[i].durationOverride) || 5;
        }
        timelineTime += Math.min(elapsed, nodeDuration);
        
        state.currentTime = timelineTime;
        updateTimelineUI();

        // Check if node is complete
        if (elapsed >= nodeDuration - 0.05) {
          advanceNode();
          return;
        }

        rafId = requestAnimationFrame(tick);
      };

      rafId = requestAnimationFrame(tick);
      elements.statusText.textContent = `Playing node ${index + 1}`;
    }

    // Advance to next node
    function advanceNode() {
      if (state.activeIndex < state.nodes.length - 1) {
        state.activeIndex++;
        playActive();
      } else {
        stopAll();
        elements.statusText.textContent = 'Timeline complete';
      }
    }

    // Stop playback
    function stopAll() {
      state.playing = false;
      state.stopRequested = true;
      
      if (rafId) {
        cancelAnimationFrame(rafId);
        rafId = null;
      }
      
      elements.baseVideo.pause();
      overlayVideos.forEach(v => v.pause());
      overlayAudios.forEach(a => a.pause());
      
      state.activeIndex = 0;
      state.currentTime = 0;
      updateTimelineUI();
      renderNodeTrack();
      
      elements.statusText.textContent = 'Stopped';
    }

    // Pause playback
    function pauseAll() {
      state.playing = false;
      elements.baseVideo.pause();
      overlayVideos.forEach(v => v.pause());
      overlayAudios.forEach(a => a.pause());
      elements.statusText.textContent = 'Paused';
    }

    // Load timeline data
    function loadTimeline() {
      const params = new URLSearchParams(window.location.search);
      const data = params.get('data');
      const name = params.get('name');

    // Timeline scrubbing (touch/click)
    let isScrubbing = false;

    function scrubToPosition(e) {
      const rect = elements.timelineTrack.getBoundingClientRect();
      const x = (e.touches ? e.touches[0].clientX : e.clientX) - rect.left;
      const percent = Math.max(0, Math.min(1, x / rect.width));
      const targetTime = percent * state.totalDuration;
      
      // Find which node this corresponds to
      let accumulatedTime = 0;
      for (let i = 0; i < state.nodes.length; i++) {
        const nodeDuration = parseFloat(state.nodes[i].durationOverride) || 5;
        if (targetTime <= accumulatedTime + nodeDuration) {
          jumpToNode(i);
          break;
        }
        accumulatedTime += nodeDuration;
      }
    }

    elements.timelineTrack.addEventListener('mousedown', (e) => {
      isScrubbing = true;
      scrubToPosition(e);

    elements.timelineTrack.addEventListener('touchstart', (e) => {
      isScrubbing = true;
      scrubToPosition(e);
    });

    document.addEventListener('mousemove', (e) => {

    });

    document.addEventListener('touchmove', (e) => {
      if (isScrubbing) scrubToPosition(e);
    });

    document.addEventListener('mouseup', () => {
      isScrubbing = false;
    });

    document.addEventListener('touchend', () => {
      isScrubbing = false;
